#include <iostream>
#include <memory>
#include <string>
#include <stdexcept>

template <typename T>
struct Node {
    T data;
    std::shared_ptr<Node<T>> next;
    std::weak_ptr<Node<T>> prev;

    Node(T value) : data(value), next(nullptr), prev() {}
};

struct Item {
    std::string name;
    int quantity;

    Item(std::string n, int q) : name(n), quantity(q) {}

    bool operator==(const Item& other) const {
        return name == other.name;
    }
};

std::ostream& operator<<(std::ostream& os, const Item& item) {
    os << item.name << " x" << item.quantity;
    return os;
}

template <typename T>
class SinglyLinkedList {
private:
    std::shared_ptr<Node<T>> head;
    int size;

public:
    SinglyLinkedList() : head(nullptr), size(0) {}

    void push_front(T value) {
        auto node = std::make_shared<Node<T>>(value);
        node->next = head;
        head = node;
        size++;
    }

    void push_back(T value) {
        auto node = std::make_shared<Node<T>>(value);

        if (!head) head = node;
        else {
            auto temp = head;
            while (temp->next)
                temp = temp->next;
            temp->next = node;
        }

        size++;
    }

    void pop_front() {
        if (!head) throw std::out_of_range("Empty");
        head = head->next;
        size--;
    }

    void pop_back() {
        if (!head) throw std::out_of_range("Empty");

        if (!head->next) head = nullptr;
        else {
            auto temp = head;
            while (temp->next->next)
                temp = temp->next;
            temp->next = nullptr;
        }

        size--;
    }

    T at(int index) {
        if (index < 0 || index >= size)
            throw std::out_of_range("Index");

        auto temp = head;
        for (int i = 0; i < index; i++)
            temp = temp->next;

        return temp->data;
    }

    void insert(int index, T value) {
        if (index < 0 || index > size)
            throw std::out_of_range("Index");

        if (index == 0) {
            push_front(value);
            return;
        }

        auto temp = head;
        for (int i = 0; i < index - 1; i++)
            temp = temp->next;

        auto node = std::make_shared<Node<T>>(value);
        node->next = temp->next;
        temp->next = node;

        size++;
    }

    void remove(int index) {
        if (index < 0 || index >= size)
            throw std::out_of_range("Index");

        if (index == 0) {
            pop_front();
            return;
        }

        auto temp = head;
        for (int i = 0; i < index - 1; i++)
            temp = temp->next;

        temp->next = temp->next->next;
        size--;
    }

    int getSize() {
        return size;
    }

    bool isEmpty() {
        return size == 0;
    }

    int find(T value) {
        auto temp = head;
        int i = 0;

        while (temp) {
            if (temp->data == value)
                return i;

            temp = temp->next;
            i++;
        }

        return -1;
    }

    void print() {
        auto temp = head;

        while (temp) {
            std::cout << temp->data << " -> ";
            temp = temp->next;
        }

        std::cout << "null\n";
    }
};

template <typename T>
class DoublyLinkedList {
private:
    std::shared_ptr<Node<T>> head;
    std::shared_ptr<Node<T>> tail;
    int size;

public:
    DoublyLinkedList() : head(nullptr), tail(nullptr), size(0) {}

    void push_front(T value) {
        auto node = std::make_shared<Node<T>>(value);

        if (!head) {
            head = tail = node;
        } else {
            node->next = head;
            head->prev = node;
            head = node;
        }

        size++;
    }

    void push_back(T value) {
        auto node = std::make_shared<Node<T>>(value);

        if (!tail) {
            head = tail = node;
        } else {
            tail->next = node;
            node->prev = tail;
            tail = node;
        }

        size++;
    }

    void pop_front() {
        if (!head) throw std::out_of_range("Empty");

        head = head->next;

        if (head)
            head->prev.reset();
        else
            tail = nullptr;

        size--;
    }

    void pop_back() {
        if (!tail) throw std::out_of_range("Empty");

        tail = tail->prev.lock();

        if (tail)
            tail->next = nullptr;
        else
            head = nullptr;

        size--;
    }

    T at(int index) {
        if (index < 0 || index >= size)
            throw std::out_of_range("Index");

        auto temp = head;

        for (int i = 0; i < index; i++)
            temp = temp->next;

        return temp->data;
    }

    void insert(int index, T value) {
        if (index < 0 || index > size)
            throw std::out_of_range("Index");

        if (index == 0) {
            push_front(value);
            return;
        }

        if (index == size) {
            push_back(value);
            return;
        }

        auto temp = head;

        for (int i = 0; i < index - 1; i++)
            temp = temp->next;

        auto node = std::make_shared<Node<T>>(value);

        node->next = temp->next;
        node->prev = temp;

        temp->next->prev = node;
        temp->next = node;

        size++;
    }

    void remove(int index) {
        if (index < 0 || index >= size)
            throw std::out_of_range("Index");

        if (index == 0) {
            pop_front();
            return;
        }

        if (index == size - 1) {
            pop_back();
            return;
        }

        auto temp = head;

        for (int i = 0; i < index; i++)
            temp = temp->next;

        auto prevNode = temp->prev.lock();
        auto nextNode = temp->next;

        prevNode->next = nextNode;
        nextNode->prev = prevNode;

        size--;
    }

    int getSize() {
        return size;
    }

    bool isEmpty() {
        return size == 0;
    }

    int find(T value) {
        auto temp = head;
        int i = 0;

        while (temp) {
            if (temp->data == value)
                return i;

            temp = temp->next;
            i++;
        }

        return -1;
    }

    void print() {
        auto temp = head;

        while (temp) {
            std::cout << temp->data << " <-> ";
            temp = temp->next;
        }

        std::cout << "null\n";
    }
};
